    + cb0[21].y * clipWGradient;

    ? clipPosition.w
    : (clipPosition.w < 0.0 ? -1e-6 : 1e-6);
  float invW2 = 1.0 / (safeW * safeW);
  float3 ndcXGradient = (clipXGradient * safeW - clipPosition.x * clipWGradient) * invW2;
  float3 ndcYGradient = (clipYGradient * safeW - clipPosition.y * clipWGradient) * invW2;

  float3 rightCandidate = cross(ndcYGradient, clipWGradient);
  float3 upCandidate = cross(clipWGradient, ndcXGradient);
  float rightLengthSquared = dot(rightCandidate, rightCandidate);
  float upLengthSquared = dot(upCandidate, upCandidate);

  if (rightLengthSquared <= 1e-14 || upLengthSquared <= 1e-14)
  {
    float aa = dot(ndcXGradient, ndcXGradient);
    float ab = dot(ndcXGradient, ndcYGradient);
    float bb = dot(ndcYGradient, ndcYGradient);
    float determinant = aa * bb - ab * ab;
    if (determinant <= 1e-14)
    {
      screenRight = float3(1.0, 0.0, 0.0);
      screenUp = float3(0.0, 1.0, 0.0);
      return;
    }

    rightCandidate = (bb * ndcXGradient - ab * ndcYGradient) / determinant;
    upCandidate = (aa * ndcYGradient - ab * ndcXGradient) / determinant;
  }

  if (dot(rightCandidate, ndcXGradient) < 0.0)
    rightCandidate = -rightCandidate;
  if (dot(upCandidate, ndcYGradient) < 0.0)
    upCandidate = -upCandidate;
  screenRight = SafeJiggleForgeNormalize(rightCandidate, float3(1.0, 0.0, 0.0));
  screenUp = SafeJiggleForgeNormalize(upCandidate, float3(0.0, 1.0, 0.0));
}
